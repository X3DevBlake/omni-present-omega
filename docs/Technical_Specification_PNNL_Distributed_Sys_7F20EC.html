pub struct DotRing {
    pub dots: [(u64, u64); N],
    pub len: usize,
}
impl DotRing {
    #[inline(always)]
    pub fn merge(&mut self, other: &Self) {
        for i in 0..other.len {
            let (node, dot) = other.dots[i];
            if let Some(pos) = self.dots[..self.len].iter().position(|&(n, _)| n == node) {
                self.dots[pos].1 = self.dots[pos].1.max(dot);
            } else if self.len < N {
                self.dots[self.len] = (node, dot);
                self.len += 1;
            }
        }
    }
}
